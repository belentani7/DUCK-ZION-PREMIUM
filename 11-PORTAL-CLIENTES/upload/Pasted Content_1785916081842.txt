const JWT_SECRET = process.env.JWT_SECRET || 'clientportal-secret-key-change-me';

const generateToken = (user) => {
  return jwt.sign(
    { id: user.id, email: user.email, role: user.role },
    JWT_SECRET,
    { expiresIn: '7d' }
  );
};

const verifyToken = (req, res, next) => {
  const authHeader = req.headers.authorization;
  if (!authHeader) {
    return res.status(401).json({ error: 'No se proporcionó token de autenticación' });
  }
  const token = authHeader.split(' ')[1];
  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    req.user = decoded;
    next();
  } catch (error) {
    return res.status(401).json({ error: 'Token inválido o expirado' });
  }
};

app.post('/api/auth/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    if (!email || !password) {
      return res.status(400).json({ error: 'Email y contraseña son obligatorios' });
    }

    const user = await prisma.user.findUnique({
      where: { email },
      include: { 
        projectsAsAdmin: true,
        projects: true
      }
    });

    if (!user) {
      return res.status(401).json({ error: 'Credenciales inválidas' });
    }

    const validPassword = await bcrypt.compare(password, user.password);
    if (!validPassword) {
      return res.status(401).json({ error: 'Credenciales inválidas' });
    }

    // Obtener datos del cliente si es CLIENT
    let clientData = null;
    if (user.role === 'CLIENT') {
      clientData = await prisma.client.findUnique({
        where: { userId: user.id },
        include: {
          projects: true,
          invoices: true
        }
      });
    }

    const token = generateToken(user);
    res.json({
      token,
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        role: user.role,
        company: user.company,
        avatar: user.avatar,
        client: clientData
      }
    });
  } catch (error) {
    console.error('Login error:', error);
    res.status(500).json({ error: 'Error interno del servidor' });
  }
});

app.post('/api/auth/register', async (req, res) => {
  try {
    const { email, password, name, role, company } = req.body;
    if (!email || !password || !name) {
      return res.status(400).json({ error: 'Email, contraseña y nombre son obligatorios' });
    }

    const existingUser = await prisma.user.findUnique({ where: { email } });
    if (existingUser) {
      return res.status(400).json({ error: 'El email ya está registrado' });
    }

    const hashedPassword = await bcrypt.hash(password, 10);
    const user = await prisma.user.create({
      data: {
        email,
        password: hashedPassword,
        name,
        role: role || 'CLIENT',
        company: company || ''
      }
    });

    // Si es CLIENT, crear registro en la tabla Client
    if (user.role === 'CLIENT') {
      await prisma.client.create({
        data: {
          userId: user.id,
          company: company || user.name + ' Empresa',
          contactName: name,
          contactEmail: email
        }
      });
    }

    const token = generateToken(user);
    res.status(201).json({
      token,
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        role: user.role,
        company: user.company
      }
    });
  } catch (error) {
    console.error('Register error:', error);
    res.status(500).json({ error: 'Error interno del servidor' });
  }
});

      const client = await prisma.client.findUnique({
        where: { userId: req.user.id }
      });
      if (client) {
        where.clientId = client.id;
      } else {
        return res.json([]);
      }
    }

    const invoices = await prisma.invoice.findMany({
      where,
      include: {
        client: { include: { user: true } },
        project: true,
        items: true
      },
      orderBy: { issueDate: 'desc' }
    });
    res.json(invoices);
  } catch (error) {
    console.error('Error invoices:', error);
    res.status(500).json({ error: 'Error al obtener facturas' });
  }
});

app.post('/api/invoices', verifyToken, async (req, res) => {
  try {
    if (req.user.role !== 'ADMIN') {
      return res.status(403).json({ error: 'Solo administradores pueden crear facturas' });
    }

    const { clientId, projectId, items, issueDate, dueDate, notes, status } = req.body;

    if (!items || items.length === 0) {
      return res.status(400).json({ error: 'La factura debe tener al menos una línea' });
    }

    // Calcular totales
    const itemsWithTotal = items.map(item => ({
      ...item,
      total: item.quantity * item.unitPrice
    }));

    const subtotal = itemsWithTotal.reduce((sum, item) => sum + item.total, 0);
    const tax = subtotal * 0.21; // IVA 21%
    const total = subtotal + tax;

    // Generar número de factura
    const count = await prisma.invoice.count();
    const number = `FAC-${String(count + 1).padStart(6, '0')}-${new Date().getFullYear()}`;

    const invoice = await prisma.invoice.create({
      data: {
        number,
        clientId,
        projectId: projectId || null,
        amount: subtotal,
        tax,
        total,
        issueDate: new Date(issueDate || Date.now()),
        dueDate: new Date(dueDate || Date.now() + 30*24*60*60*1000),
        notes,
        status: status || 'PENDING',
        items: {
          create: itemsWithTotal
        }
      },
      include: {
        client: { include: { user: true } },
        project: true,
        items: true
      }
    });
    res.status(201).json(invoice);
  } catch (error) {
    console.error('Error create invoice:', error);
    res.status(500).json({ error: 'Error al crear la factura' });
  }
});

app.get('/api/messages', verifyToken, async (req, res) => {
  try {
    const { withUser } = req.query;
    let where = {};

    if (withUser) {
      where = {
        OR: [
          { senderId: req.user.id, recipientId: withUser },
          { senderId: withUser, recipientId: req.user.id }
        ]
      };
    } else {
      // Obtener todos los contactos con los que ha chateado
      const messages = await prisma.message.findMany({
        where: {
          OR: [
            { senderId: req.user.id },
            { recipientId: req.user.id }
          ]
        },
        distinct: ['senderId', 'recipientId'],
        orderBy: { createdAt: 'desc' }
      });
      
      // Agrupar por contacto único
      const contacts = new Map();
      for (const msg of messages) {
        const contactId = msg.senderId === req.user.id ? msg.recipientId : msg.senderId;
        if (!contacts.has(contactId)) {
          contacts.set(contactId, {
            userId: contactId,
            lastMessage: msg.content,
            lastMessageAt: msg.createdAt,
            unread: msg.recipientId === req.user.id && !msg.read
          });
        }
      }
      return res.json(Array.from(contacts.values()));
    }

    const messages = await prisma.message.findMany({
      where,
      include: {
        sender: { select: { id: true, name: true, avatar: true, email: true } },
        recipient: { select: { id: true, name: true, avatar: true, email: true } }
      },
      orderBy: { createdAt: 'asc' }
    });
    res.json(messages);
  } catch (error) {
    console.error('Error messages:', error);
    res.status(500).json({ error: 'Error al obtener mensajes' });
  }
});

app.get('/api/stats', verifyToken, async (req, res) => {
  try {
    if (req.user.role !== 'ADMIN') {
      return res.status(403).json({ error: 'Solo administradores pueden ver estadísticas' });
    }

    const [
      clientCount,
      projectCount,
      deliverableCount,
      invoiceCount,
      invoiceTotal,
      overdueInvoices,
      pendingDeliverables
    ] = await Promise.all([
      prisma.client.count(),
      prisma.project.count(),
      prisma.deliverable.count(),
      prisma.invoice.count(),
      prisma.invoice.aggregate({ _sum: { total: true } }),
      prisma.invoice.count({ where: { status: 'OVERDUE' } }),
      prisma.deliverable.count({ where: { status: { in: ['PENDING', 'IN_PROGRESS'] } } })
    ]);

    // Proyectos por estado
    const projectsByStatus = await prisma.project.groupBy({
      by: ['status'],
      _count: true
    });

    // Facturas por estado
    const invoicesByStatus = await prisma.invoice.groupBy({
      by: ['status'],
      _count: true,
      _sum: { total: true }
    });

    // Ingresos mensuales (últimos 6 meses)
    const sixMonthsAgo = new Date();
    sixMonthsAgo.setMonth(sixMonthsAgo.getMonth() - 6);

    const monthlyRevenue = await prisma.invoice.groupBy({
      by: [{ month: { issueDate: true } }],
      where: {
        status: 'PAID',
        issueDate: { gte: sixMonthsAgo }
      },
      _sum: { total: true }
    });

    res.json({
      clients: clientCount,
      projects: projectCount,
      deliverables: deliverableCount,
      invoices: invoiceCount,
      totalRevenue: invoiceTotal._sum.total || 0,
      overdueInvoices,
      pendingDeliverables,
      projectsByStatus,
      invoicesByStatus,
      monthlyRevenue
    });
  } catch (error) {
    console.error('Error stats:', error);
    res.status(500).json({ error: 'Error al obtener estadísticas' });
  }
});

io.on('connection', (socket) => {
  console.log('Usuario conectado:', socket.id);

  socket.on('join', (userId) => {
    socket.join(`user:${userId}`);
    console.log(`Usuario ${userId} unido a su sala`);
  });

  socket.on('send_message', async (data) => {
    try {
      const { recipientId, content, senderId } = data;
      
      // Guardar en DB
      const message = await prisma.message.create({
        data: {
          content,
          senderId,
          recipientId,
          read: false,
          threadId: [senderId, recipientId].sort().join('-')
        },
        include: {
          sender: { select: { id: true, name: true, avatar: true } },
          recipient: { select: { id: true, name: true, avatar: true } }
        }
      });

      // Emitir al destinatario
      io.to(`user:${recipientId}`).emit('new_message', message);
      
      // Emitir al emisor (confirmación)
      io.to(`user:${senderId}`).emit('message_sent', message);
    } catch (error) {
      console.error('Error en socket send_message:', error);
    }
  });

  socket.on('mark_read', async (data) => {
    try {
      const { senderId, recipientId } = data;
      await prisma.message.updateMany({
        where: {
          senderId,
          recipientId,
          read: false
        },
        data: { read: true, readAt: new Date() }
      });
      io.to(`user:${senderId}`).emit('messages_read', { by: recipientId });
    } catch (error) {
      console.error('Error en socket mark_read:', error);
    }
  });

  socket.on('disconnect', () => {
    console.log('Usuario desconectado:', socket.id);
  });
});

function LoginScreen({ navigation }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  const handleLogin = async () => {
    if (!email || !password) {
      setError('Completa todos los campos');
      return;
    }
    setLoading(true);
    setError('');

    try {
      const response = await fetch(`${API_URL}/api/auth/login`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      });
      const data = await response.json();

      if (response.ok) {
        await AsyncStorage.setItem('token', data.token);
        await AsyncStorage.setItem('user', JSON.stringify(data.user));
        navigation.replace('Dashboard');
      } else {
        setError(data.error || 'Credenciales inválidas');
      }
    } catch (err) {
      setError('Error de conexión al servidor');
    } finally {
      setLoading(false);
    }
  };

  return (
    <SafeAreaView style={styles.loginContainer}>
      <StatusBar barStyle="light-content" backgroundColor="#0f172a" />
      <View style={styles.loginCard}>
        <Icon name="account-circle" size={80} color="#3b82f6" style={styles.loginIcon} />
        <Text style={styles.loginTitle}>ClientPortal Pro</Text>
        <Text style={styles.loginSubtitle}>Inicia sesión en tu portal de clientes</Text>

        {error ? <Text style={styles.errorText}>{error}</Text> : null}

        <TextInput
          style={styles.input}
          placeholder="Email"
          placeholderTextColor="#64748b"
          value={email}
          onChangeText={setEmail}
          autoCapitalize="none"
          keyboardType="email-address"
        />
        <TextInput
          style={styles.input}
          placeholder="Contraseña"
          placeholderTextColor="#64748b"
          value={password}
          onChangeText={setPassword}
          secureTextEntry
        />

        <TouchableOpacity
          style={[styles.loginButton, loading && styles.loginButtonDisabled]}
          onPress={handleLogin}
          disabled={loading}
        >
          {loading ? (
            <ActivityIndicator color="white" />
          ) : (
            <Text style={styles.loginButtonText}>Iniciar Sesión</Text>
          )}
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}

              <p className="text-gray-400">{selectedInvoice.client?.user?.name}</p>
            </div>
            <StatusBadge status={selectedInvoice.status} />
          </div>

          <div className="mt-6 grid grid-cols-2 md:grid-cols-4 gap-4">
            <div>
              <p className="text-sm text-gray-400">Subtotal</p>
              <p className="text-white">${selectedInvoice.amount?.toFixed(2)}</p>
            </div>
            <div>
              <p className="text-sm text-gray-400">IVA</p>
              <p className="text-white">${selectedInvoice.tax?.toFixed(2)}</p>
            </div>
            <div>
              <p className="text-sm text-gray-400">Total</p>
              <p className="text-2xl font-bold text-green-400">${selectedInvoice.total?.toFixed(2)}</p>
            </div>
            <div>
              <p className="text-sm text-gray-400">Fecha de vencimiento</p>
              <p className="text-white">{new Date(selectedInvoice.dueDate).toLocaleDateString()}</p>
            </div>
          </div>

          {selectedInvoice.items?.length > 0 && (
            <div className="mt-6">
              <h4 className="text-sm font-semibold text-gray-300 mb-2">Líneas de factura</h4>
              <div className="space-y-2">
                {selectedInvoice.items.map((item, i) => (
                  <div key={i} className="flex justify-between p-2 bg-gray-700/50 rounded-lg">
                    <span className="text-white">{item.description}</span>
                    <span className="text-gray-300">
                      {item.quantity} × ${item.unitPrice} = ${item.total}
                    </span>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      </div>
    );
  };

  const renderNav = () => {
    const adminNav = [
      { id: 'dashboard', label: 'Dashboard', icon: '📊' },
      { id: 'clients', label: 'Clientes', icon: '👥' },
      { id: 'projects', label: 'Proyectos', icon: '📂' },
      { id: 'deliverables', label: 'Entregables', icon: '📦' },
      { id: 'invoices', label: 'Facturas', icon: '💰' },
      { id: 'messages', label: 'Mensajes', icon: '💬' },
    ];

    const clientNav = [
      { id: 'dashboard', label: 'Dashboard', icon: '📊' },
      { id: 'projects', label: 'Mis Proyectos', icon: '📂' },
      { id: 'invoices', label: 'Mis Facturas', icon: '💰' },
      { id: 'messages', label: 'Chat', icon: '💬' },
    ];

    const navItems = isAdmin ? adminNav : clientNav;

    return (
      <nav className="space-y-1">
        {navItems.map(item => (
          <button
            key={item.id}
            onClick={() => {
              setView(item.id);
              // Si vamos a mensajes, cargar mensajes
              if (item.id === 'messages') loadMessages();
            }}
            className={`w-full flex items-center gap-3 px-4 py-3 rounded-lg text-sm transition ${
              view === item.id
                ? 'bg-blue-600 text-white'
                : 'text-gray-400 hover:text-white hover:bg-gray-800'
            }`}
          >
            <span className="text-xl">{item.icon}</span>
            {item.label}
          </button>
        ))}
      </nav>
    );
  };

  // ============================
  // LOADING
  // ============================
  if (loading) {
    return (
      <div className="min-h-screen bg-[#0f172a] flex items-center justify-center">
        <div className="text-center">
          <div className="inline-block w-12 h-12 border-4 border-blue-600 border-t-transparent rounded-full animate-spin" />
          <p className="text-gray-400 mt-4">Cargando ClientPortal Pro...</p>
        </div>
      </div>
    );
  }

  // ============================
  // LOGIN
  // ============================
  if (!user || !token) {
    return (
      <div className="min-h-screen bg-[#0f172a] flex items-center justify-center p-4">
        <div className="bg-gray-900 rounded-2xl p-8 w-full max-w-md border border-gray-700">
          <div className="text-center">
            <h1 className="text-3xl font-bold text-white">ClientPortal Pro</h1>
            <p className="text-gray-400 mt-2">Portal de Automatización de Clientes</p>
          </div>

          <form onSubmit={handleLogin} className="mt-8 space-y-4">
            {loginError && (
              <div className="bg-red-900/30 border border-red-800 text-red-300 p-3 rounded-lg text-sm">
                {loginError}
              </div>
            )}

            <input
              type="email"
              placeholder="Email"
              value={loginEmail}
              onChange={(e) => setLoginEmail(e.target.value)}
              className="w-full bg-gray-800 border border-gray-700 rounded-lg px-4 py-3 text-white placeholder-gray-500 focus:border-blue-500 outline-none"
            />

            <input
              type="password"
              placeholder="Contraseña"
              value={loginPassword}
              onChange={(e) => setLoginPassword(e.target.value)}
              className="w-full bg-gray-800 border border-gray-700 rounded-lg px-4 py-3 text-white placeholder-gray-500 focus:border-blue-500 outline-none"
            />

            <button
              type="submit"
              className="w-full bg-blue-600 hover:bg-blue-700 py-3 rounded-lg text-white font-semibold transition"
            >
              Iniciar Sesión
            </button>
          </form>

          <p className="text-center text-gray-400 text-sm mt-4">
            Demo: admin@portal.com / Admin123!
          </p>
        </div>
      </div>
    );
  }

  // ============================
  // MAIN APP
  // ============================
  return (
    <div className="min-h-screen bg-[#0f172a] flex">
      {/* Sidebar */}
      <aside className="w-64 bg-gray-900 border-r border-gray-800 p-4 flex flex-col">
        <div className="flex items-center gap-3 mb-8">
          <div className="w-10 h-10 bg-blue-600 rounded-lg flex items-center justify-center text-white font-bold text-xl">
            CP
          </div>
          <div>
            <p className="text-white font-semibold">ClientPortal</p>
            <p className="text-xs text-gray-400">Pro</p>
          </div>
        </div>

        {renderNav()}

        <div className="mt-auto pt-4 border-t border-gray-800">
          <div className="flex items-center gap-3">
            <div className="w-9 h-9 bg-gray-700 rounded-full flex items-center justify-center text-white font-semibold">
              {user?.name?.[0] || 'U'}
            </div>
            <div className="flex-1 min-w-0">
              <p className="text-sm text-white truncate">{user?.name}</p>
              <p className="text-xs text-gray-400">{user?.role === 'ADMIN' ? 'Admin' : 'Cliente'}</p>
            </div>
            <button
              onClick={() => {
                localStorage.removeItem('token');
                localStorage.removeItem('user');
                setUser(null);
                setToken(null);
                if (socket) socket.disconnect();
              }}
              className="text-gray-400 hover:text-white"
            >
              <span className="text-xl">🚪</span>
            </button>
          </div>
        </div>
      </aside>

      {/* Main content */}
      <main className="flex-1 p-6 overflow-y-auto">
        {view === 'dashboard' && renderDashboard()}
        {view === 'clients' && renderClients()}
        {view === 'projects' && renderProjects()}
        {view === 'deliverables' && renderDeliverables()}
        {view === 'invoices' && renderInvoices()}
        {view === 'messages' && renderMessages()}
        {view === 'projectDetail' && renderProjectDetail()}
        {view === 'invoiceDetail' && renderInvoiceDetail()}
      </main>
    </div>
  );
}
