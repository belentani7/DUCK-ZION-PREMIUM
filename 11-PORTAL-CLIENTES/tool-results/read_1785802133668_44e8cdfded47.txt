    37→// ============ API HELPER ============
    38→async function api(path: string, options?: RequestInit) {
    39→  const token = usePortalStore.getState().token;
    40→  const res = await fetch(path, {
    41→    ...options,
    42→    headers: {
    43→      'Content-Type': 'application/json',
    44→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    45→      ...options?.headers,
    46→    },
    47→  });
    48→  if (!res.ok) {
    49→    const err = await res.json().catch(() => ({ error: 'Error de servidor' }));
    50→    throw new Error(err.error || 'Error de servidor');
    51→  }
    52→  return res.json();
    53→}

    71→// ============ FORMAT HELPERS ============
    72→function formatCurrency(n: number) {
    73→  return new Intl.NumberFormat('es-ES', { style: 'currency', currency: 'EUR' }).format(n);
    74→}
    75→function formatDate(d: string) {
    76→  return new Date(d).toLocaleDateString('es-ES', { day: 'numeric', month: 'short', year: 'numeric' });
    77→}
    78→function formatTime(d: string) {
    79→  return new Date(d).toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' });
    80→}
    81→function timeAgo(d: string) {
    82→  const diff = Date.now() - new Date(d).getTime();
    83→  const mins = Math.floor(diff / 60000);
    84→  if (mins < 60) return `hace ${mins}min`;
    85→  const hours = Math.floor(mins / 60);
    86→  if (hours < 24) return `hace ${hours}h`;
    87→  const days = Math.floor(hours / 24);
    88→  return `hace ${days}d`;
    89→}

  1044→export default function HomePage() {

  1046→  const [initializing, setInitializing] = useState(true);
  1047→

  1049→  useEffect(() => {
  1050→    const restore = async () => {
  1051→      const savedToken = localStorage.getItem('portal_token');
  1052→      if (!savedToken) {
  1053→        setInitializing(false);
  1054→        return;
  1055→      }
  1056→      try {
  1057→        const userData = await api('/api/auth/me');
  1058→        if (userData) {
  1059→          setToken(savedToken);
  1060→          setUser(userData);
  1061→          initSocket(userData.id);
  1062→        }
  1063→      } catch {
  1064→        logout();
  1065→      } finally {
  1066→        setInitializing(false);
  1067→      }
  1068→    };
  1069→    restore();
  1070→  }, []);
  1071→
  1072→  // Fetch data when user is set
  1073→  useEffect(() => {
  1074→    if (!user) return;
  1075→
  1076→    const fetchData = async () => {
  1077→      try {
  1078→        const [statsData] = await Promise.all([
  1079→          api('/api/stats'),

  1083→        console.error('Error fetching stats:', err);
  1084→      }
  1085→    };
  1086→    fetchData();
  1087→  }, [user]);
  1088→
  1089→  // Fetch contextual data based on view
  1090→  useEffect(() => {
  1091→    if (!user) return;
  1092→
  1093→    const isAdmin = user.role === 'admin';

  1095→      try {

  1098→            api('/api/clients'),

  1101→          ]);
  1102→          setClients(c);
  1103→          setProjects(p);
  1104→          setInvoices(inv);

  1106→          const [p, inv, statsData] = await Promise.all([
  1107→            api('/api/projects'),

  1109→            api('/api/stats'),
  1110→          ]);

  1112→          setInvoices(inv);

  1114→        }
  1115→      } catch (err) {
  1116→        console.error('Error fetching data:', err);
  1117→      }
  1118→    };
  1119→    fetchCtx();
  1120→  }, [user?.id]);
  1121→
  1122→  if (initializing) {
  1123→    return (
  1124→      <div className="min-h-screen flex items-center justify-center">
  1125→        <div className="text-center">
  1126→          <div className="w-10 h-10 bg-emerald-600 rounded-xl flex items-center justify-center mx-auto mb-4 animate-pulse">
  1127→            <Sparkles className="w-5 h-5 text-white" />
