    46→    46→    46→async function api(path: string, options?: RequestInit) {
    47→    47→    47→  const token = usePortalStore.getState().token;
    48→    48→    48→  const res = await fetch(path, {
    49→    49→    49→    ...options,
    50→    50→    50→    headers: {
    51→    51→    51→      'Content-Type': 'application/json',
    52→    52→    52→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    53→    53→    53→      ...options?.headers,
    54→    54→    54→    },
    55→    55→    55→  });
    56→    56→    56→  if (!res.ok) {
    57→    57→    57→    const err = await res.json().catch(() => ({ error: 'Error de servidor' }));
    58→    58→    58→    throw new Error(err.error || 'Error de servidor');
    59→    59→    59→  }
    60→    60→    60→  return res.json();
    61→    61→    61→}

    80→    80→    80→function formatCurrency(n: number) {
    81→    81→    81→  return new Intl.NumberFormat('es-ES', { style: 'currency', currency: 'EUR' }).format(n);
    82→    82→    82→}
    83→    83→    83→function formatDate(d: string) {
    84→    84→    84→  return new Date(d).toLocaleDateString('es-ES', { day: 'numeric', month: 'short', year: 'numeric' });
    85→    85→    85→}
    86→    86→    86→function formatTime(d: string) {
    87→    87→    87→  return new Date(d).toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' });
    88→    88→    88→}
    89→    89→    89→function timeAgo(d: string) {
    90→    90→    90→  const diff = Date.now() - new Date(d).getTime();
    91→    91→    91→  const mins = Math.floor(diff / 60000);
    92→    92→    92→  if (mins < 60) return `hace ${mins}min`;
    93→    93→    93→  const hours = Math.floor(mins / 60);
    94→    94→    94→  if (hours < 24) return `hace ${hours}h`;
    95→    95→    95→  const days = Math.floor(hours / 24);
    96→    96→    96→  return `hace ${days}d`;
    97→    97→    97→}
