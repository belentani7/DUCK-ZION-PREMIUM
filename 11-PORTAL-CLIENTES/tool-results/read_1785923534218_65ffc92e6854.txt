    45→// ============ API HELPER ============
    46→async function api(path: string, options?: RequestInit) {
    47→  const token = usePortalStore.getState().token;
    48→  const res = await fetch(path, {
    49→    ...options,
    50→    headers: {
    51→      'Content-Type': 'application/json',
    52→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    53→      ...options?.headers,
    54→    },
    55→  });
    56→  if (!res.ok) {
    57→    const err = await res.json().catch(() => ({ error: 'Error de servidor' }));
    58→    throw new Error(err.error || 'Error de servidor');
    59→  }
    60→  return res.json();
    61→}

    79→// ============ FORMAT HELPERS ============
    80→function formatCurrency(n: number) {
    81→  return new Intl.NumberFormat('es-ES', { style: 'currency', currency: 'EUR' }).format(n);
    82→}
    83→function formatDate(d: string) {
    84→  return new Date(d).toLocaleDateString('es-ES', { day: 'numeric', month: 'short', year: 'numeric' });
    85→}
    86→function formatTime(d: string) {
    87→  return new Date(d).toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' });
    88→}
    89→function timeAgo(d: string) {
    90→  const diff = Date.now() - new Date(d).getTime();
    91→  const mins = Math.floor(diff / 60000);
    92→  if (mins < 60) return `hace ${mins}min`;
    93→  const hours = Math.floor(mins / 60);
    94→  if (hours < 24) return `hace ${hours}h`;
    95→  const days = Math.floor(hours / 24);
    96→  return `hace ${days}d`;
    97→}
