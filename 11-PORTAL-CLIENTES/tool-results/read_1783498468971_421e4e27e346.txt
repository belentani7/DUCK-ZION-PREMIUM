    12→-- ============================================================
    13→-- TABLAS
    14→-- ============================================================
    15→
    16→-- Extensión de perfiles de usuario
    17→create table public.profiles (
    18→  id uuid primary key references auth.users on delete cascade,
    19→  role text not null default 'client' check (role in ('admin','client')),
    20→  full_name text,
    21→  company text,
    22→  avatar_url text,
    23→  created_at timestamptz default now()

    25→
    26→-- Clientes (gestionados por admin)
    27→create table public.clients (
    28→  id uuid primary key default gen_random_uuid(),
    29→  profile_id uuid unique references public.profiles(id) on delete cascade,
    30→  company_name text not null,
    31→  contact_email text,
    32→  phone text,
    33→  notes text,
    34→  created_at timestamptz default now()
    35→);
    36→
    37→-- Proyectos
    38→create table public.projects (
    39→  id uuid primary key default gen_random_uuid(),
    40→  client_id uuid references public.clients(id) on delete cascade not null,
    41→  title text not null,
    42→  description text,
    43→  status text not null default 'Discovery' check (status in ('Discovery','In Progress','Review','Completed')),
    44→  start_date date,
    45→  target_end_date date,
    46→  created_at timestamptz default now(),
    47→  updated_at timestamptz default now()

    49→
    50→-- Entregables
    51→create table public.deliverables (
    52→  id uuid primary key default gen_random_uuid(),
    53→  project_id uuid references public.projects(id) on delete cascade not null,
    54→  title text not null,
    55→  description text,
    56→  due_date date not null,
    57→  status text default 'Pending' check (status in ('Pending','In Review','Approved','Rejected')),
    58→  created_at timestamptz default now()
    59→);
    60→
    61→-- Facturas
    62→create table public.invoices (
    63→  id uuid primary key default gen_random_uuid(),
    64→  client_id uuid references public.clients(id) on delete cascade not null,
    65→  invoice_number text not null unique,
    66→  issue_date date not null,
    67→  due_date date not null,
    68→  status text default 'Draft' check (status in ('Draft','Sent','Paid','Overdue')),
    69→  total_amount numeric(10,2) not null,
    70→  notes text,
    71→  created_at timestamptz default now()

    79→  quantity int not null default 1,
    80→  unit_price numeric(10,2) not null,
    81→  amount numeric(10,2) generated always as (quantity * unit_price) stored

    83→
    84→-- Mensajes
    85→create table public.messages (
    86→  id uuid primary key default gen_random_uuid(),
    87→  sender_id uuid references public.profiles(id) not null,
    88→  recipient_id uuid references public.profiles(id),
    89→  project_id uuid references public.projects(id) on delete set null,
    90→  content text not null,
    91→  is_read boolean default false,

   118→  select exists (
   119→    select 1 from public.profiles

   121→  );

   224→-- ============================================================
   225→-- TRIGGER: Crear perfil automáticamente al registrarse
   226→-- ============================================================
   227→create or replace function public.handle_new_user()
   228→returns trigger
   229→language plpgsql
   230→security definer
   231→as $$
   232→begin
   233→  insert into public.profiles (id, full_name, role)
   234→  values (new.id, coalesce(new.raw_user_meta_data->>'full_name', 'Usuario'), 'client');
   235→  return new;
   236→end;
   237→$$;
   238→
   239→create trigger on_auth_user_created
   240→after insert on auth.users
   241→for each row execute procedure public.handle_new_user();

   545→### 2.5 `hooks/useAuth.ts` (Contexto de autenticación)
   546→```typescript
   547→'use client';
   548→import { createContext, useContext, useEffect, useState } from 'react';
   549→import { supabase } from '@/lib/supabaseClient';
   550→import { Profile } from '@/lib/types';
   551→import { useRouter } from 'next/navigation';
   552→
   553→type AuthContextType = {
   554→  profile: Profile | null;
   555→  loading: boolean;
   556→  signOut: () => Promise<void>;
   557→};
   558→
   559→const AuthContext = createContext<AuthContextType>({ profile: null, loading: true, signOut: async () => {} });
   560→
   561→export function AuthProvider({ children }: { children: React.ReactNode }) {
   562→  const [profile, setProfile] = useState<Profile | null>(null);
   563→  const [loading, setLoading] = useState(true);
   564→  const router = useRouter();
   565→
   566→  useEffect(() => {
   567→    const getUser = async () => {
   568→      const { data: { user } } = await supabase.auth.getUser();
   569→      if (user) {
   570→        const { data } = await supabase.from('profiles').select('*').eq('id', user.id).single();
   571→        setProfile(data as Profile);
   572→      }
   573→      setLoading(false);
   574→    };
   575→    getUser();
   576→
   577→    const { data: authListener } = supabase.auth.onAuthStateChange((_event, session) => {
   578→      if (session?.user) {
   579→        supabase.from('profiles').select('*').eq('id', session.user.id).single().then(({ data }) => setProfile(data as Profile));
   580→      } else {
   581→        setProfile(null);
   582→      }
   583→    });
   584→    return () => authListener.subscription.unsubscribe();
   585→  }, []);
   586→
   587→  const signOut = async () => {
   588→    await supabase.auth.signOut();
   589→    setProfile(null);
   590→    router.push('/login');
   591→  };
   592→
   593→  return <AuthContext.Provider value={{ profile, loading, signOut }}>{children}</AuthContext.Provider>;
   594→}
   595→
   596→export const useAuth = () => useContext(AuthContext);
   597→```

   800→### 2.11 Chat en tiempo real (`app/dashboard/messages/page.tsx` y `app/admin/messages/page.tsx`)
   801→Utilizaré un componente `MessageThread` genérico.
   802→
   803→**`components/MessageThread.tsx`** (simplificado):
   804→```tsx
   805→'use client';
   806→import { useEffect, useState, useRef } from 'react';
   807→import { supabase } from '@/lib/supabaseClient';
   808→import { Message, Profile } from '@/lib/types';
   809→import { useAuth } from '@/hooks/useAuth';
   810→import { Button } from './ui/button';
   811→import { Input } from './ui/input';
   812→
   813→type Props = {
   814→  recipientId: string;
   815→  projectId?: string;
   816→};
   817→
   818→export default function MessageThread({ recipientId, projectId }: Props) {
   819→  const { profile } = useAuth();
   820→  const [messages, setMessages] = useState<Message[]>([]);
   821→  const [newMsg, setNewMsg] = useState('');
   822→  const bottomRef = useRef<HTMLDivElement>(null);
   823→
   824→  useEffect(() => {
   825→    if (!profile) return;
   826→    // Cargar mensajes iniciales
   827→    const fetchMessages = async () => {
   828→      const { data } = await supabase.from('messages')
   829→        .select('*, sender:profiles!sender_id(full_name)')
   830→        .or(`and(sender_id.eq.${profile.id},recipient_id.eq.${recipientId}),and(sender_id.eq.${recipientId},recipient_id.eq.${profile.id})`)
   831→        .order('created_at');
   832→      setMessages(data as Message[]);
   833→    };
   834→    fetchMessages();
   835→
   836→    // Suscripción en tiempo real
   837→    const channel = supabase
   838→      .channel('messages-room')
   839→      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'messages' }, (payload) => {
   840→        const newMsg = payload.new as Message;
   841→        if (
   842→          (newMsg.sender_id === profile.id && newMsg.recipient_id === recipientId) ||
   843→          (newMsg.sender_id === recipientId && newMsg.recipient_id === profile.id)
   844→        ) {
   845→          setMessages(prev => [...prev, newMsg]);
   846→        }
   847→      })
   848→      .subscribe();
   849→
   850→    return () => { supabase.removeChannel(channel); };
   851→  }, [profile, recipientId]);
   852→
   853→  useEffect(() => {
   854→    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
   855→  }, [messages]);
   856→
   857→  const sendMessage = async () => {
   858→    if (!newMsg.trim() || !profile) return;
   859→    await supabase.from('messages').insert({
   860→      sender_id: profile.id,
   861→      recipient_id: recipientId,
   862→      project_id: projectId,
   863→      content: newMsg,
   864→    });
   865→    setNewMsg('');
   866→  };
   867→
   868→  return (
   869→    <div className="flex flex-col h-[calc(100vh-12rem)]">
   870→      <div className="flex-1 overflow-y-auto space-y-2 p-2">
   871→        {messages.map(msg => (
   872→          <div key={msg.id} className={`flex ${msg.sender_id === profile?.id ? 'justify-end' : 'justify-start'}`}>
   873→            <div className={`max-w-[70%] p-3 rounded-lg ${msg.sender_id === profile?.id ? 'bg-indigo-100' : 'bg-gray-200'}`}>
   874→              <p className="text-sm">{msg.content}</p>
   875→              <span className="text-xs text-gray-500">{new Date(msg.created_at).toLocaleTimeString()}</span>
   876→            </div>
   877→          </div>
   878→        ))}
   879→        <div ref={bottomRef} />
   880→      </div>
   881→      <div className="flex gap-2 p-2 border-t">
   882→        <Input value={newMsg} onChange={e => setNewMsg(e.target.value)} placeholder="Escribe un mensaje..." onKeyDown={e => e.key === 'Enter' && sendMessage()} />
   883→        <Button onClick={sendMessage}>Enviar</Button>
   884→      </div>
   885→    </div>
   886→  );
   887→}
   888→```

   895→'use client';

   903→
   904→const invoiceSchema = z.object({
   905→  client_id: z.string().uuid(),
   906→  invoice_number: z.string().min(1),
   907→  issue_date: z.string(),
   908→  due_date: z.string(),
   909→  status: z.enum(['Draft','Sent','Paid','Overdue']),
   910→  notes: z.string().optional(),
   911→  items: z.array(z.object({
   912→    description: z.string().min(1),
   913→    quantity: z.number().min(1),
   914→    unit_price: z.number().min(0),
   915→  })).min(1),
   916→});
   917→
   918→type FormData = z.infer<typeof invoiceSchema>;
   919→
   920→export default function CreateInvoiceForm({ clients }: { clients: {id:string, company_name:string}[] }) {
   921→  const { register, handleSubmit, control, formState: { errors } } = useForm<FormData>({
   922→    resolver: zodResolver(invoiceSchema),
   923→    defaultValues: { status: 'Draft', items: [{ description: '', quantity: 1, unit_price: 0 }] }
   924→  });
   925→  const { fields, append, remove } = useFieldArray({ control, name: 'items' });
   926→
   927→  const onSubmit = async (data: FormData) => {
   928→    // Calcular total
   929→    const total = data.items.reduce((sum, item) => sum + item.quantity * item.unit_price, 0);
   930→    const { data: invoice, error } = await supabase.from('invoices').insert({
   931→      client_id: data.client_id,
   932→      invoice_number: data.invoice_number,
   933→      issue_date: data.issue_date,
   934→      due_date: data.due_date,
   935→      status: data.status,
   936→      total_amount: total,
   937→      notes: data.notes
   938→    }).select().single();
   939→    if (error) { toast.error(error.message); return; }
   940→
   941→    // Insertar items
   942→    const itemsToInsert = data.items.map(item => ({
   943→      invoice_id: invoice.id,
   944→      ...item
   945→    }));
   946→    const { error: itemsError } = await supabase.from('invoice_items').insert(itemsToInsert);
   947→    if (itemsError) toast.error(itemsError.message);
   948→    else toast.success('Factura creada');
   949→  };
   950→
   951→  return (
   952→    <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
   953→      <select {...register('client_id')} className="border p-2 rounded w-full">
   954→        <option value="">Seleccionar cliente</option>
   955→        {clients.map(c => <option key={c.id} value={c.id}>{c.company_name}</option>)}
   956→      </select>
   957→      <Input {...register('invoice_number')} placeholder="Número factura" />
   958→      <div className="grid grid-cols-2 gap-4">
   959→        <Input type="date" {...register('issue_date')} />
   960→        <Input type="date" {...register('due_date')} />
   961→      </div>
   962→      <select {...register('status')} className="border p-2 rounded">
   963→        <option value="Draft">Borrador</option>
   964→        <option value="Sent">Enviada</option>
   965→        <option value="Paid">Pagada</option>
   966→        <option value="Overdue">Vencida</option>
   967→      </select>
   968→      <div>
   969→        <h3 className="font-medium">Líneas de factura</h3>
   970→        {fields.map((field, index) => (
   971→          <div key={field.id} className="flex gap-2 mt-2">
   972→            <Input placeholder="Descripción" {...register(`items.${index}.description`)} />
   973→            <Input type="number" placeholder="Cant." {...register(`items.${index}.quantity`, { valueAsNumber: true })} />
   974→            <Input type="number" step="0.01" placeholder="Precio" {...register(`items.${index}.unit_price`, { valueAsNumber: true })} />
   975→            <Button type="button" variant="destructive" onClick={() => remove(index)}>X</Button>
   976→          </div>

  1076→### Row Level Security (RLS) Policies
  1077→Apply these policies after table creation:
  1078→
  1079→- **profiles**: Users can select their own profile. Admin can select all.
  1080→- **clients**: 
  1081→  - Client: SELECT only where `profile_id = auth.uid()`. No INSERT/UPDATE/DELETE.
  1082→  - Admin: full CRUD.
  1083→- **projects**:
  1084→  - Client: SELECT only where `client_id` matches the client record of the logged-in profile.
  1085→  - Admin: full CRUD.
  1086→- **deliverables**:
  1087→  - Client: SELECT only if they own the parent project.
  1088→  - Admin: full CRUD.
  1089→- **invoices** & **invoice_items**:
  1090→  - Client: SELECT only if `client_id` matches.
  1091→  - Admin: full CRUD.
  1092→- **messages**:
  1093→  - Sender (client or admin) can INSERT with `sender_id = auth.uid()` and recipient according to logic. 
  1094→  - Clients can SELECT messages where sender_id or recipient_id matches their profile id.
  1095→  - Admin can SELECT all messages.
  1096→  - Update (mark read) allowed by recipient.

  1189→## 1. Procesos en segundo plano (Background Jobs)
  1190→
  1191→Supabase ofrece dos caminos limpios:
  1192→
  1193→- **pg_cron** (extensión de PostgreSQL) → Ideal para tareas recurrentes simples (cada minuto, hora, día).

  1196→### 1.1 Actualización automática de facturas vencidas
  1197→
  1198→Queremos que las facturas con `status != 'Paid'` y `due_date < today()` pasen automáticamente a `'Overdue'`. Lo hará un trabajo diario.
  1199→
  1200→#### Paso 1 – Habilitar pg_cron en Supabase
  1201→En el dashboard de Supabase, ve a `Database > Extensions` y activa `pg_cron`.  
  1202→(Si no está disponible, usaremos una Edge Function programada).
  1203→
  1204→#### Paso 2 – Crear la función que actualiza
  1205→Ejecuta en el SQL Editor:
  1206→
  1207→```sql
  1208→create or replace function public.auto_mark_overdue_invoices()
  1209→returns void
  1210→language sql
  1211→as $$
  1212→  update public.invoices
  1213→  set status = 'Overdue'
  1214→  where status in ('Draft', 'Sent')
  1215→    and due_date < current_date;
  1216→$$;
  1217→```
  1218→
  1219→#### Paso 3 – Programar la tarea
  1220→```sql
  1221→select cron.schedule(
  1222→  'mark-overdue-invoices',  -- nombre único
  1223→  '0 3 * * *',              -- todos los días a las 03:00 UTC
  1224→  'select public.auto_mark_overdue_invoices();'
  1225→);
  1226→```
  1227→
  1228→Ya tienes un proceso automático que revisa las facturas cada noche y las marca como vencidas.

  1259→## 2. Auto‑regulación y auto‑recuperación ante errores
  1260→
  1261→La “auto‑regulación” significa que el sistema se mantiene estable y se repara solo. Lo conseguimos con:
  1262→
  1263→- **Restricciones en la base de datos** (CHECK, NOT NULL, claves foráneas) → Ya las tienes.
  1264→- **Triggers que corrigen inconsistencias** (ej. recalcular total de factura si se modifica un item).
  1265→- **Reintentos automáticos en el frontend**.
  1266→- **Reconexión automática de suscripciones realtime**.
  1267→
  1268→### 2.1 Trigger para mantener la integridad del total de la factura
  1269→
  1270→Aunque `invoice_items.amount` es una columna generada, podríamos necesitar que `invoices.total_amount` se actualice si se añade, modifica o elimina una línea. Para eso creamos un trigger:
  1271→
  1272→```sql
  1273→-- Función que recalcula el total de una factura
  1274→create or replace function public.update_invoice_total()
  1275→returns trigger
  1276→language plpgsql
  1277→as $$
  1278→begin
  1279→  if TG_OP = 'DELETE' then
  1280→    update public.invoices
  1281→    set total_amount = (select coalesce(sum(amount),0) from public.invoice_items where invoice_id = OLD.invoice_id)
  1282→    where id = OLD.invoice_id;
  1283→  else
  1284→    update public.invoices
  1285→    set total_amount = (select coalesce(sum(amount),0) from public.invoice_items where invoice_id = NEW.invoice_id)
  1286→    where id = NEW.invoice_id;
  1287→  end if;
  1288→  return null;
  1289→end;
  1290→$$;
  1291→
  1292→-- Asignar el trigger a la tabla invoice_items
  1293→create trigger invoice_items_total_update
  1294→after insert or update or delete on public.invoice_items
  1295→for each row execute function public.update_invoice_total();
  1296→```
  1297→
  1298→Así, si alguien manipula las líneas directamente, el total se ajusta solo.
  1299→
  1300→### 2.2 Auto‑recuperación en el frontend (reintentos y reconexión)
  1301→
  1302→**Reintentos en llamadas a la API**  
  1303→Puedes envolver tus funciones de `lib/api.ts` con una estrategia simple de reintento:
  1304→
  1305→```typescript
  1306→// lib/apiUtils.ts
  1307→export async function withRetry<T>(fn: () => Promise<T>, retries = 3): Promise<T> {
  1308→  for (let i = 0; i < retries; i++) {
  1309→    try {
  1310→      return await fn();
  1311→    } catch (error) {
  1312→      if (i === retries - 1) throw error;
  1313→      console.warn(`Retry ${i+1} after error:`, error);
  1314→      await new Promise(r => setTimeout(r, 1000 * Math.pow(2, i))); // backoff
  1315→    }
  1316→  }
  1317→  throw new Error('Unreachable');
  1318→}
  1319→```
  1320→
  1321→Luego en `api.ts`:
  1322→```typescript
  1323→import { withRetry } from './apiUtils';
  1324→
  1325→export async function getClientProjects(clientId: string) {
  1326→  return withRetry(() => supabase.from('projects').select('*').eq('client_id', clientId).order(...));
  1327→}
  1328→```
  1329→
  1330→**Reconexión automática de canales realtime**  
  1331→El cliente de Supabase reintenta la conexión automáticamente, pero puedes añadir un listener para reiniciar el canal si falla:
  1332→
  1333→```typescript
  1334→const channel = supabase.channel('room', {
  1335→  config: { broadcast: { ack: true } },
  1336→}).subscribe((status) => {
  1337→  if (status === 'CLOSED' || status === 'CHANNEL_ERROR') {
  1338→    console.log('Canal cerrado, reconectando...');
  1339→    setTimeout(() => channel.subscribe(), 2000);
  1340→  }
  1341→});
  1342→```

  1348→## 4. Orquestación con Edge Functions programadas (alternativa sin pg_cron)
  1349→
  1350→Si tu plan de Supabase no incluye pg_cron, crea una Edge Function que se ejecute periódicamente mediante un **Cron Job de Supabase** (disponible en el dashboard en `Edge Functions > Cron Jobs`).
  1351→
  1352→Ejemplo de función `update-overdue`:
  1353→
  1354→```typescript
  1355→// supabase/functions/update-overdue/index.ts
  1356→import { serve } from 'https://deno.land/std@0.168.0/http/server.ts';
  1357→import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';
  1358→
  1359→serve(async (_req) => {
  1360→  const supabase = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);
  1361→  const { error } = await supabase
  1362→    .from('invoices')
  1363→    .update({ status: 'Overdue' })
  1364→    .in('status', ['Draft', 'Sent'])
  1365→    .lt('due_date', new Date().toISOString().split('T')[0]);
  1366→  if (error) return new Response(JSON.stringify(error), { status: 500 });
  1367→  return new Response(JSON.stringify({ success: true }), { headers: { 'Content-Type': 'application/json' } });
  1368→});
  1369→```

  1371→Luego en el dashboard programa el cron: `0 3 * * *`.
