    37→// ============ API HELPER ============
    38→async function api(path: string, options?: RequestInit) {
    39→  const token = usePortalStore.getState().token;
    40→  const res = await fetch(path, {
    41→    ...options,
    42→    headers: {
    43→      'Content-Type': 'application/json',
    44→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    45→      ...options?.headers,
    46→    },
    47→  });
    48→  if (!res.ok) {
    49→    const err = await res.json().catch(() => ({ error: 'Error de servidor' }));
    50→    throw new Error(err.error || 'Error de servidor');
    51→  }
    52→  return res.json();
    53→}

    71→// ============ FORMAT HELPERS ============
    72→function formatCurrency(n: number) {
    73→  return new Intl.NumberFormat('es-ES', { style: 'currency', currency: 'EUR' }).format(n);
    74→}
    75→function formatDate(d: string) {
    76→  return new Date(d).toLocaleDateString('es-ES', { day: 'numeric', month: 'short', year: 'numeric' });
    77→}
    78→function formatTime(d: string) {
    79→  return new Date(d).toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' });
    80→}
    81→function timeAgo(d: string) {
    82→  const diff = Date.now() - new Date(d).getTime();
    83→  const mins = Math.floor(diff / 60000);
    84→  if (mins < 60) return `hace ${mins}min`;
    85→  const hours = Math.floor(mins / 60);
    86→  if (hours < 24) return `hace ${hours}h`;
    87→  const days = Math.floor(hours / 24);
    88→  return `hace ${days}d`;
    89→}

  1059→      const savedToken = localStorage.getItem('portal_token');
  1060→      if (!savedToken) {
  1061→        setInitializing(false);
  1062→        return;
  1063→      }
  1064→      try {
  1065→        const userData = await api('/api/auth/me');

  1067→          setToken(savedToken);

  1069→          initSocket(userData.id);

  1072→        logout();
  1073→      } finally {
  1074→        setInitializing(false);
  1075→      }
  1076→    };
  1077→    restore();
  1078→  }, []);
  1079→
  1080→  // Fetch data when user is set
  1081→  useEffect(() => {
  1082→    if (!user) return;
  1083→

  1086→        const [statsData] = await Promise.all([
  1087→          api('/api/stats'),

  1092→      }

  1096→
  1097→  // Fetch contextual data based on view

  1122→        }
  1123→      } catch (err) {
  1124→        console.error('Error fetching data:', err);
  1125→      }
  1126→    };
  1127→    fetchCtx();
  1128→  }, [user?.id]);
  1129→
  1130→  if (initializing) {
